============================================================
Scenario A — Minimal Cluster (no addons, 1 node group)
============================================================

# root/main.tf
module "eks" {
  source = "../modules/eks"

  cluster_name     = var.cluster_name     # "demo-cluster"
  cluster_version  = var.cluster_version  # "1.30"
  vpc_id           = var.vpc_id           # "vpc-xxxx"
  private_subnets  = var.private_subnets  # ["subnet-aaa","subnet-bbb"]

  public_access_enabled  = var.public_access_enabled  # true
  public_access_cidrs    = var.public_access_cidrs    # ["0.0.0.0/0"]

  EKS_NODE_GROUP_DEFINITIONS = var.node_groups
}

# root/terraform.tfvars
cluster_name     = "demo-cluster"
cluster_version  = "1.30"
vpc_id           = "vpc-0123456789abcdef0"
private_subnets  = ["subnet-1111","subnet-2222"]

public_access_enabled = true
public_access_cidrs   = ["0.0.0.0/0"]

node_groups = [
  {
    type             = "ON_DEMAND"
    suffix           = "general"
    fleet            = ["t3.medium"]
    ami              = "ami-012345"
    root_volume_size = 20
    desired_size     = 2
    max_size         = 3
    min_size         = 1
    is_taint         = false
    taint            = null
  }


####### this part to be able for authintcate to the eks through local ########

authentication_mode = "API" #OR CONFIG_MAP 

eks_access_roles = {
  "access1" = { 
    ### this only for type=standard the default it make association for the access entry ###
    principal_arn     = "role arn"
    kubernetes_groups = ["masters"]
    policy_arn        = "policy arn"
    access_scope_type = "cluster" #OR NAMESPACE
  }
  "access2" = { 
    #### this for any type except STANDARD as the association is disabled on this case ###
    principal_arn     = "role arn"
    type              = EC2_LINUX
    access_scope_type = "cluster" #OR NAMESPACE
  }

}

]

Scenario D — Full Production Cluster (Addons + IRSA + Tags)
============================================================

# root/main.tf
module "eks" {
  source = "../modules/eks"

  cluster_name     = var.cluster_name     # "prod-platform"
  cluster_version  = var.cluster_version  # "1.30"
  vpc_id           = var.vpc_id           # "vpc-xxxx"
  private_subnets  = var.private_subnets  # ["subnet-aaa","subnet-bbb"]
  public_subnets   = var.public_subnets   # ["subnet-ccc","subnet-ddd"]
  eks_kms_arn      = var.eks_kms_arn      # "arn:aws:kms:region:acct:key/xxxx"
  DEFAULT_TAGS     = var.default_tags     # { Project="Core", Env="prod" }

  private_access_allowed = var.private_access_allowed # true
  public_access_enabled  = var.public_access_enabled  # true
  public_access_cidrs    = var.public_access_cidrs    # ["0.0.0.0/0"]

  irsa_bindings = var.irsa_bindings # EBS & EFS roles

  enable_vpc_cni     = var.enable_vpc_cni     # true
  vpc_cni_version    = var.vpc_cni_version    # "v1.16.4-eksbuild.1"
  enable_coredns     = var.enable_coredns     # true
  coredns_version    = var.coredns_version    # "v1.11.1-eksbuild.2"
  enable_kube_proxy  = var.enable_kube_proxy  # true
  kube_proxy_version = var.kube_proxy_version # "v1.30.2-eksbuild.1"
  enable_ebs_csi     = var.enable_ebs_csi     # true
  ebs_csi_version    = var.ebs_csi_version    # "v1.30.0-eksbuild.1"

  EKS_NODE_GROUP_DEFINITIONS = var.node_groups
}
